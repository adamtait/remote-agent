# cws CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `cws`, a Go CLI binary that provisions and manages Google Cloud Workstations with Claude Code, reading per-project config from `.workstation.yaml` discovered by walking up the directory tree.

**Architecture:** Walk up from `$PWD` to find `.workstation.yaml` (project config), merge with `~/.config/cws/config.yaml` (global user config), then shell out to `gcloud` for all GCP operations. Cobra handles CLI structure; subcommands share config via package-level vars set in `PersistentPreRunE`.

**Tech Stack:** Go 1.22+, `github.com/spf13/cobra`, `gopkg.in/yaml.v3`, `golang.org/x/term`

package config

import (
	"fmt"
	"os"
	"path/filepath"
)

// FindProjectConfig walks up from startDir looking for .workstation.yaml,
// stopping when it reaches stopDir or the filesystem root.
func FindProjectConfig(startDir, stopDir string) (string, error) {
	dir := startDir
	for {
		candidate := filepath.Join(dir, ".workstation.yaml")
		if _, err := os.Stat(candidate); err == nil {
			return candidate, nil
		}
		if dir == stopDir || dir == "/" {
			break
		}
		parent := filepath.Dir(dir)
		if parent == dir {
			break
		}
		dir = parent
	}
	return "", fmt.Errorf("no .workstation.yaml found; create one in your project root (see README)")
}
```

package config

import (
	"errors"
	"os"
	"path/filepath"

	"gopkg.in/yaml.v3"
)

// GlobalConfigPath returns the default path for the global user config.
func GlobalConfigPath() string {
	home, _ := os.UserHomeDir()
	return filepath.Join(home, ".config", "cws", "config.yaml")
}

// LoadGlobalConfig reads and parses the global config file.
// Returns an empty GlobalConfig (no error) if the file does not exist.
func LoadGlobalConfig(path string) (*GlobalConfig, error) {
	data, err := os.ReadFile(path)
	if errors.Is(err, os.ErrNotExist) {
		return &GlobalConfig{}, nil
	}
	if err != nil {
		return nil, err
	}
	var cfg GlobalConfig
	if err := yaml.Unmarshal(data, &cfg); err != nil {
		return nil, err
	}
	return &cfg, nil
}

package config

import (
	"os"
	"regexp"
	"strings"

	"gopkg.in/yaml.v3"
)

// LoadProjectConfig reads and parses a .workstation.yaml file.
func LoadProjectConfig(path string) (*ProjectConfig, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}
	var cfg ProjectConfig
	if err := yaml.Unmarshal(data, &cfg); err != nil {
		return nil, err
	}
	return &cfg, nil
}

// Merge combines a project config and global config into a single Config,
// applying precedence: global project overrides > project config > global defaults > hardcoded defaults.
func Merge(project *ProjectConfig, global *GlobalConfig) (*Config, error) {
	cfg := &Config{}

	// Hardcoded defaults
	cfg.MachineType = "e2-standard-4"
	cfg.DiskSizeGB = 50
	cfg.ImageTag = "latest"
	cfg.ClusterName = "claude-dev-cluster"
	cfg.ConfigName = "claude-dev-config"
	cfg.ServiceAccountName = "claude-workstation-sa"
	cfg.ImageName = "claude-code-workstation"
	cfg.RepoName = "claude-workstation-images"

	// Global defaults
	if global.Defaults.Region != "" {
		cfg.Region = global.Defaults.Region
	}
	cfg.GitUserName = global.Defaults.GitUserName
	cfg.GitUserEmail = global.Defaults.GitUserEmail

	// Project config
	if project.ProjectID != "" {
		cfg.ProjectID = project.ProjectID
	}
	if project.Region != "" {
		cfg.Region = project.Region
	}
	if project.ClusterName != "" {
		cfg.ClusterName = project.ClusterName
	}
	if project.ConfigName != "" {
		cfg.ConfigName = project.ConfigName
	}
	if project.GithubRepoURL != "" {
		cfg.GithubRepoURL = project.GithubRepoURL
	}
	if project.MachineType != "" {
		cfg.MachineType = project.MachineType
	}
	if project.DiskSizeGB != 0 {
		cfg.DiskSizeGB = project.DiskSizeGB
	}

	// Global project overrides (highest precedence)
	if override, ok := global.Projects[cfg.ProjectID]; ok {
		if override.ServiceAccountName != "" {
			cfg.ServiceAccountName = override.ServiceAccountName
		}
		if override.ImageName != "" {
			cfg.ImageName = override.ImageName
		}
		if override.ImageTag != "" {
			cfg.ImageTag = override.ImageTag
		}
		if override.RepoName != "" {
			cfg.RepoName = override.RepoName
		}
	}

	// Derive username for workstation name
	username := os.Getenv("USER")
	if username == "" {
		username = "user"
	}
	cfg.WorkstationName = WorkstationName(cfg.ProjectID, username)

	return cfg, nil
}

var nonAlphanumHyphen = regexp.MustCompile(`[^a-z0-9]+`)

// WorkstationName returns the default workstation name for a project/user pair.
// Exported so it can be tested directly.
func WorkstationName(projectID, username string) string {
	clean := func(s string) string {
		s = strings.ToLower(s)
		s = nonAlphanumHyphen.ReplaceAllString(s, "-")
		s = strings.Trim(s, "-")
		return s
	}
	name := clean(projectID) + "-" + clean(username)
	if len(name) > 63 {
		name = name[:63]
	}
	name = strings.TrimRight(name, "-")
	return name
}

package gcloud

import (
	"bytes"
	"os"
	"os/exec"
)

// Runner executes gcloud subcommands.
type Runner interface {
	// Run executes gcloud with args, streaming stdout/stderr to the terminal.
	Run(args ...string) error
	// Output executes gcloud with args and returns stdout bytes. Stderr streams to terminal.
	Output(args ...string) ([]byte, error)
	// Check executes gcloud with args, discarding all output. Returns true if exit code is 0.
	Check(args ...string) bool
}

// RealRunner shells out to the gcloud binary on PATH.
type RealRunner struct{}

func (r RealRunner) Run(args ...string) error {
	cmd := exec.Command("gcloud", args...)
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	cmd.Stdin = os.Stdin
	return cmd.Run()
}

func (r RealRunner) Output(args ...string) ([]byte, error) {
	cmd := exec.Command("gcloud", args...)
	var buf bytes.Buffer
	cmd.Stdout = &buf
	cmd.Stderr = os.Stderr
	if err := cmd.Run(); err != nil {
		return nil, err
	}
	return buf.Bytes(), nil
}

func (r RealRunner) Check(args ...string) bool {
	cmd := exec.Command("gcloud", args...)
	return cmd.Run() == nil
}
```

package gcloud

import (
	"fmt"
	"strconv"
	"strings"
)

// WorkstationInfo holds the parsed fields from a gcloud workstations describe/list call.
type WorkstationInfo struct {
	Name  string
	State string
	Host  string
}

// Client wraps a Runner with typed gcloud workstation operations.
type Client struct {
	runner Runner
}

// NewClient creates a Client using the provided Runner.
func NewClient(r Runner) *Client {
	return &Client{runner: r}
}

// List returns all workstations in the given cluster (across all configs).
func (c *Client) List(project, region, cluster string) ([]WorkstationInfo, error) {
	out, err := c.runner.Output(
		"workstations", "list",
		"--cluster", cluster,
		"--region", region,
		"--project", project,
		"--format", "value(name,state)",
	)
	if err != nil {
		return nil, err
	}
	var result []WorkstationInfo
	for _, line := range strings.Split(strings.TrimSpace(string(out)), "\n") {
		if line == "" {
			continue
		}
		parts := strings.Split(line, "\t")
		info := WorkstationInfo{}
		if len(parts) > 0 {
			info.Name = parts[0]
		}
		if len(parts) > 1 {
			info.State = parts[1]
		}
		result = append(result, info)
	}
	return result, nil
}

// Describe returns details for a specific workstation.
func (c *Client) Describe(project, region, cluster, cfgName, name string) (*WorkstationInfo, error) {
	out, err := c.runner.Output(
		"workstations", "describe", name,
		"--cluster", cluster,
		"--config", cfgName,
		"--region", region,
		"--project", project,
		"--format", "value(name,state,host)",
	)
	if err != nil {
		return nil, err
	}
	line := strings.TrimSpace(string(out))
	parts := strings.Split(line, "\t")
	info := &WorkstationInfo{}
	if len(parts) > 0 {
		info.Name = parts[0]
	}
	if len(parts) > 1 {
		info.State = parts[1]
	}
	if len(parts) > 2 {
		info.Host = parts[2]
	}
	return info, nil
}

// Exists returns true if the workstation can be described without error.
func (c *Client) Exists(project, region, cluster, cfgName, name string) bool {
	return c.runner.Check(
		"workstations", "describe", name,
		"--cluster", cluster, "--config", cfgName,
		"--region", region, "--project", project,
	)
}

// Start starts a stopped workstation.
func (c *Client) Start(project, region, cluster, cfgName, name string) error {
	return c.runner.Run(
		"workstations", "start", name,
		"--cluster", cluster, "--config", cfgName,
		"--region", region, "--project", project,
	)
}

// Stop stops a running workstation.
func (c *Client) Stop(project, region, cluster, cfgName, name string) error {
	return c.runner.Run(
		"workstations", "stop", name,
		"--cluster", cluster, "--config", cfgName,
		"--region", region, "--project", project,
	)
}

// Delete deletes a workstation (caller must confirm first).
func (c *Client) Delete(project, region, cluster, cfgName, name string) error {
	return c.runner.Run(
		"workstations", "delete", name,
		"--cluster", cluster, "--config", cfgName,
		"--region", region, "--project", project,
		"--quiet",
	)
}

// SSH opens an interactive SSH session to the workstation.
func (c *Client) SSH(project, region, cluster, cfgName, name string) error {
	return c.runner.Run(
		"workstations", "ssh", name,
		"--cluster", cluster, "--config", cfgName,
		"--region", region, "--project", project,
	)
}

// Logs streams or snapshots container logs for the workstation from Cloud Logging.
// If follow is true, streams indefinitely (until Ctrl-C). If tail > 0, limits to that many entries.
func (c *Client) Logs(project, name string, follow bool, tail int) error {
	filter := fmt.Sprintf(
		`resource.type="workstations.googleapis.com/Workstation" AND resource.labels.workstation_id=%q`,
		name,
	)
	args := []string{"logging", "read", filter, "--project", project}
	if follow {
		args = append(args, "--follow")
	}
	if tail > 0 {
		args = append(args, "--limit", strconv.Itoa(tail))
	}
	return c.runner.Run(args...)

package cmd

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"

	"github.com/adamtait/remote-agent/cws/internal/config"
	"github.com/adamtait/remote-agent/cws/internal/gcloud"
)

// resolveWorkstationName returns the workstation name to target.
// If name is non-empty, it is used directly. Otherwise, the default (<project_id>-<username>)
// is tried; if not found and exactly one workstation exists, that one is used; if multiple
// exist, the user is prompted to choose.
func resolveWorkstationName(cfg *config.Config, client *gcloud.Client, name string) (string, error) {
	if name != "" {
		return name, nil
	}

	// Try the default workstation name.
	if client.Exists(cfg.ProjectID, cfg.Region, cfg.ClusterName, cfg.ConfigName, cfg.WorkstationName) {
		info, err := client.Describe(cfg.ProjectID, cfg.Region, cfg.ClusterName, cfg.ConfigName, cfg.WorkstationName)
		if err == nil {
			return info.Name, nil
		}
	}

	// Fall back: list all workstations in the cluster.
	workstations, err := client.List(cfg.ProjectID, cfg.Region, cfg.ClusterName)
	if err != nil {
		return "", fmt.Errorf("listing workstations: %w", err)
	}
	if len(workstations) == 0 {
		return "", fmt.Errorf("no workstations found in cluster %q", cfg.ClusterName)
	}
	if len(workstations) == 1 {
		return workstations[0].Name, nil
	}

	// Multiple workstations — prompt.
	fmt.Fprintln(os.Stderr, "Multiple workstations found:")
	for i, ws := range workstations {
		fmt.Fprintf(os.Stderr, "  %d) %s (%s)\n", i+1, ws.Name, ws.State)
	}
	fmt.Fprintf(os.Stderr, "Select workstation [1-%d]: ", len(workstations))
	scanner := bufio.NewScanner(os.Stdin)
	if !scanner.Scan() {
		return "", fmt.Errorf("no selection made")
	}
	n, err := strconv.Atoi(strings.TrimSpace(scanner.Text()))
	if err != nil || n < 1 || n > len(workstations) {
		return "", fmt.Errorf("invalid selection")
	}
	return workstations[n-1].Name, nil
}

// ensureClaudeSecret detects or prompts for the Claude auth secret.
// Returns (authMode, secretName, error).
func ensureClaudeSecret(r gcloud.Runner, projectID string) (string, string, error) {
	if r.Check("secrets", "describe", "claude-code-oauth-token", "--project", projectID) {
		fmt.Println("Found existing 'claude-code-oauth-token' — using Claude subscription auth.")
		return "oauth", "claude-code-oauth-token", nil
	}
	if r.Check("secrets", "describe", "anthropic-api-key", "--project", projectID) {
		fmt.Println("Found existing 'anthropic-api-key' — using API key auth.")
		return "apikey", "anthropic-api-key", nil
	}

	fmt.Print(`
Choose Claude Code authentication:
  1) Claude Pro/Max/Team OAuth token  (recommended for subscribers)
  2) Anthropic API key                (pay-as-you-go)

To generate an OAuth token: run ` + "`claude setup-token`" + ` locally.
`)
	scanner := bufio.NewScanner(os.Stdin)
	fmt.Fprint(os.Stderr, "Enter choice [1/2]: ")
	if !scanner.Scan() {
		return "", "", fmt.Errorf("no input")
	}
	choice := strings.TrimSpace(scanner.Text())

	switch choice {
	case "1":
		if err := ensureSecret(r, projectID, "claude-code-oauth-token",
			"Paste CLAUDE_CODE_OAUTH_TOKEN (sk-ant-oat01-...)"); err != nil {
			return "", "", err
		}
		return "oauth", "claude-code-oauth-token", nil
	case "2":
		if err := ensureSecret(r, projectID, "anthropic-api-key", "Enter ANTHROPIC_API_KEY"); err != nil {
			return "", "", err
		}
		return "apikey", "anthropic-api-key", nil
	default:
		return "", "", fmt.Errorf("invalid choice: %q", choice)
	}
}

// ensureSecret creates the named secret if it doesn't exist, prompting for its value.
func ensureSecret(r gcloud.Runner, projectID, name, prompt string) error {
	if r.Check("secrets", "describe", name, "--project", projectID) {
		fmt.Printf("Secret %q already exists (use 'gcloud secrets versions add' to rotate).\n", name)
		return nil
	}
	fmt.Fprintf(os.Stderr, "%s: ", prompt)
	value, err := term.ReadPassword(int(os.Stdin.Fd()))
	fmt.Fprintln(os.Stderr)
	if err != nil {
		return fmt.Errorf("reading secret value: %w", err)
	}
	if len(strings.TrimSpace(string(value))) == 0 {
		return fmt.Errorf("secret value cannot be empty")
	}

	// Pipe the value into `gcloud secrets create` via a temp file to avoid shell history.
	f, err := os.CreateTemp("", "cws-secret-*")
	if err != nil {
		return err
	}
	defer os.Remove(f.Name())
	if _, err := f.Write(value); err != nil {
		f.Close()
		return err
	}
	f.Close()

	return r.Run(
		"secrets", "create", name,
		"--replication-policy=automatic",
		"--data-file="+f.Name(),
		"--project", projectID,
	)
}
